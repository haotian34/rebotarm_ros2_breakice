cmake_minimum_required(VERSION 3.8)
project(rebotarm_rviz_panel)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rebotarm_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(${PROJECT_NAME} SHARED src/safety_panel.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} rviz_common pluginlib rclcpp std_srvs rebotarm_msgs)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()

if(BUILD_TESTING)
  enable_testing()
  add_executable(panel_smoke test/panel_smoke.cpp)
  ament_target_dependencies(panel_smoke rviz_common pluginlib rclcpp std_srvs rebotarm_msgs)
  target_link_libraries(panel_smoke Qt5::Widgets)
  add_test(NAME panel_smoke COMMAND panel_smoke)
  set_tests_properties(panel_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ROS_DOMAIN_ID=137" TIMEOUT 20)
  add_test(NAME panel_rs_namespace COMMAND panel_smoke)
  set_tests_properties(panel_rs_namespace PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ROS_DOMAIN_ID=138;REBOTARM_SAFETY_NAMESPACE=rebotarm_rs" TIMEOUT 20)
endif()
